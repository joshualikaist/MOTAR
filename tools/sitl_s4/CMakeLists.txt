cmake_minimum_required(VERSION 3.10)
project(motar_s4_bridge CXX)
set(CMAKE_CXX_STANDARD 17)
find_package(gazebo REQUIRED)
include_directories(${GAZEBO_INCLUDE_DIRS})
link_directories(${GAZEBO_LIBRARY_DIRS})
list(APPEND CMAKE_CXX_FLAGS "${GAZEBO_CXX_FLAGS}")
add_library(motar_s4_bridge SHARED motar_s4_bridge.cc)
target_link_libraries(motar_s4_bridge ${GAZEBO_LIBRARIES} pthread)
add_library(motar_rgbd_bridge SHARED motar_rgbd_bridge.cc)
target_link_libraries(motar_rgbd_bridge ${GAZEBO_LIBRARIES} pthread)
